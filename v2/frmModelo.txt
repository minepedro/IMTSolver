Private Sub Carregar()
    Dim m As Modelo, i As Long, ladoDir As String
    mN = 0: ReDim mLhs(1 To 1): ReDim mRel(1 To 1): ReDim mRhs(1 To 1)
    Me.Controls("txtObj").Text = ""
    Me.Controls("txtVars").Text = ""
    Me.Controls("txtValor").Text = "0"
    Me.Controls("optMin").Value = True
    Me.Controls("chkNaoNeg").Value = True

    If LerModelo(mWs, m) Then
        If Not m.obj Is Nothing Then Me.Controls("txtObj").Text = Endereco(m.obj)
        Select Case m.tipo
            Case TIPO_MAX: Me.Controls("optMax").Value = True
            Case TIPO_VALOR: Me.Controls("optValor").Value = True
            Case Else: Me.Controls("optMin").Value = True
        End Select
        Me.Controls("txtValor").Text = Trim$(Str$(m.valorAlvo))
        Me.Controls("txtVars").Text = Endereco(m.vars)
        Me.Controls("chkNaoNeg").Value = m.naoNeg
        For i = 1 To m.n
            ' Com IIf isto quebrava com erro 91: o VBA avalia TODOS os
            ' ramos, e numa restricao int/bin o objeto rhs vale Nothing.
            If m.r(i).rel >= REL_INT Then
                ladoDir = ""
            ElseIf m.r(i).rhsEhConst Then
                ladoDir = Trim$(Str$(m.r(i).rhsConst))
            Else
                ladoDir = Endereco(m.r(i).rhs)
            End If
            AnotaRestricao Endereco(m.r(i).lhs), m.r(i).rel, ladoDir
        Next i
    End If
    MostrarLista

    CarregarSolvers False, Opcao("solver", "HiGHS")
    Me.Controls("txtTempo").Text = Opcao("tempo", "300")
    Me.Controls("txtGap").Text = Opcao("gap", "0")
    Me.Controls("chkSens").Value = (Opcao("sens", "1") = "1")
    Me.Controls("txtEmail").Text = Opcao("email", "")
End Sub

Private Sub CarregarSolvers(ByVal forcar As Boolean, ByVal escolhido As String)
    Dim item As String
    Dim lista As Collection, p As Variant, i As Long
    With Me.Controls("cboSolver")
        .Clear
        mNSolvers = 0: ReDim mSolverNomes(1 To 1)
        Set lista = ListarSolvers(forcar)
        If lista.Count = 0 Then
            For Each p In Array("CBC", "HiGHS", "SCIP")
                mNSolvers = mNSolvers + 1: ReDim Preserve mSolverNomes(1 To mNSolvers)
                mSolverNomes(mNSolvers) = p: .AddItem p & "   (motor nao encontrado)"
            Next p
        Else
            For Each p In lista
                mNSolvers = mNSolvers + 1: ReDim Preserve mSolverNomes(1 To mNSolvers)
                mSolverNomes(mNSolvers) = p(0)
                ' com IIf isto dava erro 9: o VBA avalia os dois ramos, e
                ' p(4) nao existe numa lista guardada por uma versao antiga
                ' o tipo vem logo depois do nome: e o que decide a escolha.
                ' Antes ficava no fim e era o primeiro a ser cortado.
                item = p(1)
                If UBound(p) >= 4 Then item = item & "  -  " & p(4)
                item = item & "  -  " & p(2)
                .AddItem item
            Next p
        End If
        .ListIndex = 0
        For i = 1 To mNSolvers
            If UCase$(mSolverNomes(i)) = UCase$(escolhido) Then .ListIndex = i - 1
        Next i
    End With
End Sub

Private Function Salvar() As Boolean
    Dim m As Modelo, i As Long, r As Range, txt As String
    Set m.ws = mWs

    ' objetivo
    txt = Trim$(Me.Controls("txtObj").Text)
    If txt <> "" Then
        Set r = Faixa(txt)
        If r Is Nothing Then Avisa "Nao entendi a celula do objetivo: " & txt: Exit Function
        If r.Cells.Count <> 1 Then Avisa "O objetivo precisa ser uma celula so.": Exit Function
        If Not r.HasFormula Then Avisa "A celula do objetivo (" & txt & ") nao tem formula. Ela deveria calcular o total a otimizar.": Exit Function
        Set m.obj = r
    End If
    If Me.Controls("optMax").Value Then
        m.tipo = TIPO_MAX
    ElseIf Me.Controls("optValor").Value Then
        m.tipo = TIPO_VALOR
        m.valorAlvo = Val(Replace(Me.Controls("txtValor").Text, ",", "."))
    Else
        m.tipo = TIPO_MIN
    End If

    ' variaveis
    txt = Trim$(Me.Controls("txtVars").Text)
    If txt = "" Then Avisa "Faltam as celulas variaveis.": Exit Function
    Set r = Faixa(txt)
    If r Is Nothing Then Avisa "Nao entendi as celulas variaveis: " & txt: Exit Function
    Set m.vars = r
    m.naoNeg = Me.Controls("chkNaoNeg").Value

    ' restricoes
    m.n = mN
    If mN > 0 Then
        ReDim m.r(1 To mN)
        For i = 1 To mN
            Set r = Faixa(mLhs(i))
            If r Is Nothing Then Avisa "Restricao " & i & ": lado esquerdo invalido (" & mLhs(i) & ")": Exit Function
            Set m.r(i).lhs = r
            m.r(i).rel = mRel(i)
            If mRel(i) <= REL_GE Then
                If EhNumero(mRhs(i)) Then
                    m.r(i).rhsEhConst = True
                    m.r(i).rhsConst = Val(Replace(mRhs(i), ",", "."))
                Else
                    Set r = Faixa(mRhs(i))
                    If r Is Nothing Then Avisa "Restricao " & i & ": lado direito invalido (" & mRhs(i) & ")": Exit Function
                    Set m.r(i).rhs = r
                End If
            Else
                m.r(i).rhsEhConst = True
            End If
        Next i
    End If
    If m.obj Is Nothing And mN = 0 Then Avisa "Defina pelo menos um objetivo ou uma restricao.": Exit Function

    On Error GoTo Falhou
    GravarModelo m
    On Error GoTo 0
    ' conferir: o que foi gravado tem que voltar na leitura
    If Not TemModelo(mWs) Then
        Avisa "O modelo nao ficou gravado na aba '" & mWs.Name & "'. " & _
              "A pasta esta protegida ou e somente leitura?"
        Exit Function
    End If

    ' opcoes do usuario
    If Me.Controls("cboSolver").ListIndex >= 0 Then GravarOpcao "solver", mSolverNomes(Me.Controls("cboSolver").ListIndex + 1)
    GravarOpcao "tempo", Trim$(Str$(Val(Me.Controls("txtTempo").Text)))
    GravarOpcao "gap", Trim$(Str$(Val(Replace(Me.Controls("txtGap").Text, ",", "."))))
    GravarOpcao "sens", IIf(Me.Controls("chkSens").Value, "1", "0")
    GravarOpcao "email", Trim$(Me.Controls("txtEmail").Text)
    Salvar = True
    Exit Function
Falhou:
    Avisa "Nao consegui gravar o modelo (erro " & Err.Number & "): " & Err.Description
End Function

'--------------------------------------------------------------
' restricoes
'--------------------------------------------------------------
Private Sub AnotaRestricao(ByVal lhs As String, ByVal rel As Long, ByVal rhs As String)
    mN = mN + 1
    ReDim Preserve mLhs(1 To mN): ReDim Preserve mRel(1 To mN): ReDim Preserve mRhs(1 To mN)
    mLhs(mN) = lhs: mRel(mN) = rel: mRhs(mN) = rhs
End Sub

Private Sub MostrarLista()
    Dim i As Long, sel As Long
    sel = mLst.ListIndex
    mLst.Clear
    For i = 1 To mN
        mLst.AddItem Left$(mLhs(i) & Space$(18), 18) & "  " & Left$(SinalTexto(mRel(i)) & "   ", 4) & "  " & mRhs(i)
    Next i
    If sel >= 0 And sel < mN Then mLst.ListIndex = sel
End Sub

Private Function SinalTexto(ByVal rel As Long) As String
    Select Case rel
        Case REL_LE: SinalTexto = "<="
        Case REL_EQ: SinalTexto = "="
        Case REL_GE: SinalTexto = ">="
        Case REL_INT: SinalTexto = "int"
        Case REL_BIN: SinalTexto = "bin"
    End Select
End Function

Private Sub PainelRestricao(ByVal visivel As Boolean)
    Dim n As Variant
    For Each n In Array("lblEd", "txtLhs", "btnLhs", "cboRel", "txtRhs", "btnRhs", "btnAplicar", "btnCancRestr")
        Me.Controls(n).Visible = visivel
    Next n
    mBtnAdd.Enabled = Not visivel: mBtnEdit.Enabled = Not visivel
    mBtnDel.Enabled = Not visivel: mBtnDelAll.Enabled = Not visivel
    mBtnResolver.Enabled = Not visivel: mBtnSalvar.Enabled = Not visivel
End Sub

    Me.Controls("txtLhs").Text = mLhs(mEditando)
    Me.Controls("txtRhs").Text = mRhs(mEditando)

Private Sub mBtnDel_Click()
    Dim i As Long, k As Long
    If mLst.ListIndex < 0 Then Exit Sub
    k = mLst.ListIndex + 1
    For i = k To mN - 1
        mLhs(i) = mLhs(i + 1): mRel(i) = mRel(i + 1): mRhs(i) = mRhs(i + 1)
    Next i
    mN = mN - 1
    MostrarLista
End Sub

Private Sub mBtnDelAll_Click()
    If mN = 0 Then Exit Sub
    If MsgBox("Excluir todas as " & mN & " restricoes?", vbQuestion + vbYesNo, IMT_NOME) = vbYes Then
        mN = 0: MostrarLista
    End If
End Sub

Private Sub mCboRel_Change()
    Dim ehTipo As Boolean
    If mMontando Then Exit Sub
    ehTipo = (mCboRel.ListIndex >= 3)
    Me.Controls("txtRhs").Enabled = Not ehTipo
    mBtnRhs.Enabled = Not ehTipo
    If ehTipo Then Me.Controls("txtRhs").Text = ""
End Sub

Private Sub mBtnAplicar_Click()
    Dim lhs As String, rhs As String, rel As Long, r As Range
    lhs = Trim$(Me.Controls("txtLhs").Text)
    rhs = Trim$(Me.Controls("txtRhs").Text)
    rel = mCboRel.ListIndex + 1
    Set r = Faixa(lhs)
    If r Is Nothing Then Avisa "Lado esquerdo invalido: " & lhs: Exit Sub
    lhs = Endereco(r)
    If rel <= REL_GE Then
        If rhs = "" Then Avisa "Falta o lado direito.": Exit Sub
        If Not EhNumero(rhs) Then
            Set r = Faixa(rhs)
            If r Is Nothing Then Avisa "Lado direito invalido: " & rhs: Exit Sub
            rhs = Endereco(r)
        Else
            rhs = Trim$(Str$(Val(Replace(rhs, ",", "."))))
        End If
    Else
        rhs = ""
    End If
    If mEditando = 0 Then
        AnotaRestricao lhs, rel, rhs
    Else
        mLhs(mEditando) = lhs: mRel(mEditando) = rel: mRhs(mEditando) = rhs
    End If
    MostrarLista
    PainelRestricao False
End Sub

End Sub

'--------------------------------------------------------------
' selecao de faixas
'--------------------------------------------------------------
Private Function EscolherFaixa(ByVal prompt As String, ByVal atual As String) As String
    Dim r As Range
    EscolherFaixa = atual
    On Error Resume Next
    Set r = Application.InputBox(prompt, IMT_NOME, atual, Type:=8)
    On Error GoTo 0
    If Not r Is Nothing Then EscolherFaixa = Endereco(r)
End Function

Private Function Endereco(r As Range) As String
    ' A1:B3 quando e da aba do modelo; 'Outra'!A1 quando nao e
    If r.Worksheet Is mWs Then
        Endereco = r.Address(False, False)
    Else
        Endereco = "'" & r.Worksheet.Name & "'!" & r.Address(False, False)
    End If
End Function

Private Function Faixa(ByVal txt As String) As Range
    On Error Resume Next
    If InStr(txt, "!") > 0 Then
        ' Application.Range olha a pasta ATIVA; com duas pastas abertas
        ' isso pegava a aba de outro arquivo. Evaluate fica na pasta do modelo.
        Set Faixa = mWs.Evaluate(txt)
        If TypeName(Faixa) <> "Range" Then Set Faixa = Nothing
    Else
        Set Faixa = mWs.Range(txt)
    End If
End Function

Private Function EhNumero(ByVal txt As String) As Boolean
    EhNumero = IsNumeric(Replace(txt, ",", "."))
End Function

End Sub

Private Sub mBtnSugerir_Click()
    ' variaveis = celulas de valor numerico das quais o objetivo depende
    Dim obj As Range, prec As Range, c As Range, achadas As Range
    Set obj = Faixa(Trim$(Me.Controls("txtObj").Text))
    If obj Is Nothing Then Avisa "Informe primeiro a celula do objetivo.": Exit Sub
    On Error Resume Next
    Set prec = obj.Precedents
    On Error GoTo 0
    If prec Is Nothing Then Avisa "O objetivo nao depende de nenhuma celula desta aba.": Exit Sub
    For Each c In prec.Cells
        If Not c.HasFormula Then
            If IsNumeric(c.Value2) Or IsEmpty(c.Value2) Then
                If achadas Is Nothing Then Set achadas = c Else Set achadas = Application.Union(achadas, c)
            End If
        End If
    Next c
    If achadas Is Nothing Then
        Avisa "Nao achei celulas de valor entre as que o objetivo usa."
    Else
        Me.Controls("txtVars").Text = Endereco(achadas)
        MsgBox "Sugestao: " & Endereco(achadas) & "  (" & achadas.Cells.Count & " celulas)" & vbLf & vbLf & _
               "Confira: os coeficientes (custos, tempos) tambem sao celulas de valor e podem ter " & _
               "entrado na lista. Tire o que nao for decisao.", vbInformation, IMT_NOME
    End If
End Sub

        Me.Hide
    End If
End Sub
